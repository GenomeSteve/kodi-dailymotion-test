0365: def _dm_playlist(url, headers, ceiling):
0366:     """Resolve HLS master playlists without assuming absolute URLs or NAME tags."""
0367:     if not isinstance(url, str) or not url.startswith(('https://', 'http://')):
0368:         return None
0369:     try:
0370:         manifest = client.request(url, headers=headers)
0371:     except Exception as exc:
0372:         xbmc.log('DAILYMOTION - HLS request failed: {0}'.format(exc), xbmc.LOGWARNING)
0373:         return None
0374:     if not isinstance(manifest, str) or not manifest.strip():
0375:         return None
0376:     # A media playlist is already playable. Don't fetch its segments.
0377:     if '#EXT-X-STREAM-INF' not in manifest:
0378:         return url if '#EXTM3U' in manifest else None
0379:     variants = []
0380:     pending = None
0381:     for line in manifest.splitlines():
0382:         line = line.strip()
0383:         if line.startswith('#EXT-X-STREAM-INF:'):
0384:             attrs = line.split(':', 1)[1]
0385:             match = re.search(r'RESOLUTION=\d+x(\d+)', attrs)
0386:             name = re.search(r'NAME="?(\d+)', attrs)
0387:             bandwidth = re.search(r'BANDWIDTH=(\d+)', attrs)
0388:             pending = int(match.group(1)) if match else (int(name.group(1)) if name else 0)
0389:             if pending == 0 and bandwidth:
0390:                 pending = min(2160, max(240, int(bandwidth.group(1)) // 1500))
0391:         elif pending is not None and line and not line.startswith('#'):

0393:             pending = None

0395:         return None
